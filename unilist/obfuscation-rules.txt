# 混淆规则。
# 详见 https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/source-obfuscation
#
# 这是**组件库**，对外 API 名必须保留，否则使用者拿到的声明文件里
# 导出会变成混淆后的单字母（实测过：入口会变成 `export { i } from "./src/main/ets/d/e"`），
# 调用方直接编译不过。
#
# 因此这里**不能**开启：
#   -enable-export-obfuscation   会把导出名混淆掉
#   -enable-filename-obfuscation 会把文件名混淆掉（入口 Index 也会被改名）
#   -enable-toplevel-obfuscation 会把顶层作用域名混淆掉

# 字符串属性名（动态访问用）不能混淆，否则运行期按名字取属性会失败
-keep-property-name
  $$,
  componentName

# 保留所有顶层声明名，避免组件类名被改
-keep-global-name
  UniList,
  UniGroupList,
  UniGrid,
  UniWaterfall,
  UniSwipeItem,
  UniState,
  UniLoadState,
  UniSkeletonVariant,
  UniTheme,
  UniTexts,
  UniRes,
  UniSwipeAction,
  UniSwipeActionHandler,
  IUniSection,
  UniItemKey,
  UniFlowHeight

# 压缩空白，移除日志（保留导出名）
-enable-property-obfuscation
-compact
-remove-log
